# Core environment

jupyter  # runs interactive notebooks (.ipynb) to develop and test your code.
tqdm  # shows progress bars in loops (useful for training and data loading).
numpy  # base library for numerical operations and arrays.
pandas  # handles structured data, CSVs, and annotations.
matplotlib  # plots charts, histograms, and visualizations.
seaborn  # higher-level visualization interface built on Matplotlib.
scikit-learn  # provides metrics, data splits, and preprocessing tools.
scipy  # scientific computing library used for filters, transforms, and math utilities.
pillow  # image reading, writing, and manipulation library.
opencv-python  # advanced image operations (cropping, resizing, augmenting).
opencv-python-headless  # headless OpenCV for servers/Colab (no GUI dependencies).
pyyaml  # YAML configuration file support.
pydantic  # data validation and structured configuration models.
python-dotenv  # loads environment variables from .env files.
imageio  # reading/writing images and GIFs.
scikit-image  # image processing algorithms (filters, morphology, segmentation).

# Deep learning

torch  # main PyTorch library for tensor operations and model training.
torchvision  # model architectures (ResNet, UNet, etc.) and image transforms.
torchaudio  # audio support; required by some torchvision dependencies.
timm  # PyTorch Image Models library, adds many pretrained networks for transfer learning.
albumentations  # fast and flexible image augmentation library.
pycocotools  # used for COCO-format annotations.
segmentation-models-pytorch  # high-level segmentation architectures (UNet, DeepLabV3, etc.).
ultralytics  # YOLOv8 object detection and segmentation framework.
transformers  # HuggingFace models (SegFormer, ViT, etc.).
einops  # clean tensor rearrangement operations.

# Experiment tracking & visualization

wandb  # experiment tracking, logging, and visualization.
tensorboard  # training curves, scalars, histograms, and model graph visualization.
plotly  # interactive visualizations and dashboards.

# Utilities

rich  # makes terminal logs more readable with colors and tables.
hydra-core  # helps manage experiment configurations with clean YAML setups.

# Jupyter Notebook Support

ipywidgets  # interactive widgets inside notebooks.
jupyter-contrib-nbextensions  # extra notebook features and UI improvements.
nbformat  # notebook file format utilities.

# Google Colab Utilities

gdown  # download files from Google Drive easily.
pydrive2  # Google Drive API wrapper for uploading/downloading programmatically.

# Development Tools

black  # automatic code formatter.
flake8  # Python linter for style and error checking.
pytest  # testing framework for unit tests.
